188async def get_or_create_user(
189 db: Connection,
190 user_data: AuthUser,
191):
192 """Get user from User table if exists, else create."""
193 try:
194 upsert_sql = """
195 WITH upserted_user AS (
196 INSERT INTO users (
197 sub,
198 username,
199 email_address,
200 profile_img,
201 role,
202 registered_at
203 )
204 VALUES (
205 %(user_sub)s,
206 %(username)s,
207 %(email_address)s,
208 %(profile_img)s,
209 %(role)s,
210 NOW()
211 )
212 ON CONFLICT (sub)
213 DO UPDATE SET
214 profile_img = EXCLUDED.profile_img,
215 last_login_at = NOW()
216 RETURNING sub, username, email_address, profile_img, role
217 )
218
219 SELECT
220 u.sub,
221 u.username,
222 u.email_address,
223 u.profile_img,
224 u.role,
225
226 -- Aggregate the organisation IDs managed by the user
227 array_agg(
228 DISTINCT om.organisation_id
229 ) FILTER (WHERE om.organisation_id IS NOT NULL) AS orgs_managed,
230
231 -- Aggregate project roles for the user, as project:role pairs
232 jsonb_object_agg(
233 ur.project_id,
234 COALESCE(ur.role, 'MAPPER')
235 ) FILTER (WHERE ur.project_id IS NOT NULL) AS project_roles
236
237 FROM upserted_user u
238 LEFT JOIN user_roles ur ON u.sub = ur.user_sub
239 LEFT JOIN organisation_managers om ON u.sub = om.user_sub
240 GROUP BY
241 u.sub,
242 u.username,
243 u.email_address,
244 u.profile_img,
245 u.role;
246 """
247
248 async with db.cursor(row_factory=class_row(DbUser)) as cur:
249 await cur.execute(
250 upsert_sql,
251 {
252 "user_sub": user_data.sub,
253 "username": user_data.username,
254 "email_address": user_data.email,
255 "profile_img": user_data.profile_img or "",
256 "role": UserRole(user_data.role).name,
257 },
258 )
259 db_user_details = await cur.fetchone()
260
261 if not db_user_details:
262 raise HTTPException(
263 status_code=HTTPStatus.NOT_FOUND,
264 detail=f"User ({user_data.sub}) could not be inserted in db",
265 )
266
267 return db_user_details
268
269 except Exception as e:
270 await db.rollback()
271 log.exception(f"Exception occurred: {e}", stack_info=True)
272 raise HTTPException(status_code=HTTPStatus.BAD_REQUEST, detail=str(e)) from e

292@router.get("/refresh/management", response_model=FMTMUser)
293async def refresh_management_cookies(
294 request: Request,
295 current_user: Annotated[AuthUser, Depends(login_required)],
296):
297 """Uses the refresh token to generate a new access token.
298
299 This endpoint is specific to the management desktop frontend.
300 Any temp auth cookies will be ignored and removed.
301 Authentication is required.
302 If signed in with login method other than OSM, the user will be logged out and
303 a forbidden status will be returned.
304
305 NOTE this endpoint has no db calls and returns in ~2ms.
306 """
307 if "osm" not in current_user.sub.lower():
308 response = Response(
309 status_code=HTTPStatus.FORBIDDEN,
310 content="Please log in using OSM for management access.",
311 )
312 cookie_names = [
313 settings.cookie_name,
314 f"{settings.cookie_name}_refresh",
315 ]
316
317 response = await expire_cookies(response, cookie_names)
318 return response
319
320 return await refresh_cookies(
321 request,
322 current_user,
323 settings.cookie_name,
324 f"{settings.cookie_name}_refresh",
325 )

328@router.get("/refresh/mapper", response_model=Optional[FMTMUser])
329async def refresh_mapper_token(
330 request: Request,
331 current_user: Annotated[AuthUser, Depends(mapper_login_required)],
332):
333 """Uses the refresh token to generate a new access token.
334
335 This endpoint is specific to the mapper mobile frontend.
336 By default the user will be logged in with a temporary auth cookie.
337 Authentication is optional, if the user wishes to be attributed for contributions.
338
339 NOTE this endpoint has no db calls and returns in ~2ms.
340 """
341 try:
342 # If standard login cookie is passed, use that
343 response = await refresh_cookies(
344 request,
345 current_user,
346 settings.cookie_name,
347 f"{settings.cookie_name}_refresh",
348 )
349 return response
350 except HTTPException:
351 # NOTE we allow for token verification to fail for the main cookie
352 # and fallback to to generate a temp auth cookie
353 pass
354
355 # Refresh the temp cookies (we must re-create the 'sub' field)
356 temp_jwt_details = {
357 **current_user.model_dump(exclude=["id"]),
358 "sub": current_user.sub,
359 "aud": settings.FMTM_DOMAIN,
360 "iat": int(time()),
361 "exp": int(time()) + 86400, # set token expiry to 1 day
362 }
363
364 fmtm_token, refresh_token = create_jwt_tokens(temp_jwt_details)
365 # NOTE be sure to not append content=current_user.model_dump() to this JSONResponse
366 # as we want the login state on the frontend to remain empty (allowing the user to
367 # log in via OSM instead / override)
368 response = JSONResponse(status_code=HTTPStatus.OK, content=temp_jwt_details)
369 return set_cookies(
370 response,
371 fmtm_token,
372 refresh_token,
373 f"{settings.cookie_name}_temp",
374 f"{settings.cookie_name}_temp_refresh",
375 )
